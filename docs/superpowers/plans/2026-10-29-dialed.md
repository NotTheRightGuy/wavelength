# Dialed Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build Dialed, a mobile-first multiplayer Wavelength-style party game (teams or free play) with a Supabase backend, Motion animation, cranking motion and generated sound.

**Architecture:** A Vite/React SPA talks to Supabase only through `security definer` RPC functions (no client writes) and subscribes to Realtime for state changes plus a broadcast channel for the live needle. Pure game rules live in `src/game-engine` (TDD) and are mirrored in plpgsql; both are checked against shared fixtures. The secret target lives in an RLS-protected table.

**Tech Stack:** React 19, Vite, TypeScript, Tailwind 4, shadcn, `motion`, `@supabase/supabase-js`, `react-router`, `qrcode.react`, `vite-plugin-pwa`, Vitest, pgTAP (Supabase CLI), Playwright, pnpm.

**Spec:** `docs/superpowers/specs/2026-10-29-dialed-design.md`

## Global Constraints

- App name is **Dialed**; folder/codename is `Wavelength`. Never use "Wavelength" in user-visible copy or copy official artwork/logos.
- Board value range is 0–100 (0 = left end). Target is 20 wide, centre in 10..90, bands 2/3/4/3/2, each 4 wide.
- Dial points 4/3/2/0; left/right bonus +1; default winning score 10; default laps 2.
- Team mode 2–6 teams (one device per team); free play 3–12 players; rooms expire after 6 h idle.
- Clients never write tables directly; all writes go through RPCs. The target is never sent to any session other than the active seat before reveal.
- Needle broadcast throttled to about 20 Hz; receiving screens smooth it with a spring.
- Dial timer off by default; options 60/90/120 s; ticking in last 10 s; auto-lock at zero.
- Sounds are generated with Web Audio (no audio files). Mute and volume persist per device (`localStorage`).
- `prefers-reduced-motion` replaces spin/shutter with short fades; every drag has a button alternative.
- pnpm only. Motion is imported from `motion/react`.
- Package versions: pin exact versions at install time (`pnpm add -E`) and use releases at least two weeks old.

```ts
import { BANDS } from './constants'

export type DialPoints = 0 | 2 | 3 | 4
export type LeftRight = 'left' | 'right'

export function bandPoints(target: number, dial: number): DialPoints {
  const distance = Math.abs(target - dial)
  for (const band of BANDS) {
    if (distance <= band.maxDistance) return band.points
  }
  return 0
}

/** `guess` states where the target is relative to the dial. */
export function leftRightCorrect(target: number, dial: number, guess: LeftRight): boolean {
  if (target === dial) return false
  return guess === (target < dial ? 'left' : 'right')
}
```

```ts
import { BOARD_MAX, BOARD_MIN, TARGET_HALF_WIDTH } from './constants'

export function generateTarget(rng: () => number = Math.random): number {
  const span = BOARD_MAX - BOARD_MIN - 2 * TARGET_HALF_WIDTH
  const raw = BOARD_MIN + TARGET_HALF_WIDTH + rng() * span
  return Math.round(raw * 100) / 100
}
```

`src/game-engine/team.ts`:

```ts
import { DEFAULT_WINNING_SCORE, LR_BONUS } from './constants'

export type Phase = 'playing' | 'sudden_death' | 'finished'

export interface TeamGame {
  scores: number[]
  activeTeam: number
  winningScore: number
  phase: Phase
  sdStart: number[]
  sdPending: number[]
  sdTied: number[]
  winners: number[]
}

export interface TurnOutcome {
  dialPoints: number
  /** Indexes of teams whose left/right guess was correct. */
  bonusTeams: number[]
}

export function newTeamGame(teamCount: number, winningScore = DEFAULT_WINNING_SCORE): TeamGame {
  return {
    scores: Array(teamCount).fill(0),
    activeTeam: 0,
    winningScore,
    phase: 'playing',
    sdStart: [],
    sdPending: [],
    sdTied: [],
    winners: [],
  }
}

function enterSuddenDeath(g: TeamGame, leaders: number[]): TeamGame {
  return { ...g, phase: 'sudden_death', sdTied: leaders, sdPending: [...leaders], sdStart: [...g.scores], activeTeam: leaders[0] }
}

function advanceSuddenDeath(g: TeamGame, scores: number[]): TeamGame {
  const pending = g.sdPending.filter((t) => t !== g.activeTeam)
  if (pending.length > 0) return { ...g, scores, sdPending: pending, activeTeam: pending[0] }
  const gains = g.sdTied.map((t) => ({ t, gain: scores[t] - g.sdStart[t] }))
  const best = Math.max(...gains.map((x) => x.gain))
  const leaders = gains.filter((x) => x.gain === best).map((x) => x.t)
  if (leaders.length === 1) return { ...g, scores, phase: 'finished', winners: leaders, sdPending: [] }
  return enterSuddenDeath({ ...g, scores }, leaders)
}

export function applyTurn(g: TeamGame, o: TurnOutcome): TeamGame {
  if (g.phase === 'finished') throw new Error('game is finished')
  const scores = [...g.scores]
  scores[g.activeTeam] += o.dialPoints
  for (const t of o.bonusTeams) {
    if (t === g.activeTeam) continue
    if (g.phase === 'sudden_death' && !g.sdTied.includes(t)) continue
    scores[t] += LR_BONUS
  }
  if (g.phase === 'sudden_death') return advanceSuddenDeath(g, scores)

  const bestOther = Math.max(...scores.filter((_, i) => i !== g.activeTeam))
  if (o.dialPoints === 4 && scores[g.activeTeam] < bestOther) return { ...g, scores } // catch-up turn

  const top = Math.max(...scores)
  if (top >= g.winningScore) {
    const leaders = scores.flatMap((s, i) => (s === top ? [i] : []))
    if (leaders.length === 1) return { ...g, scores, phase: 'finished', winners: leaders }
    return enterSuddenDeath({ ...g, scores }, leaders)
  }
  return { ...g, scores, activeTeam: (g.activeTeam + 1) % scores.length }
}
```

```ts
export function psychicScore(guesserPoints: number[]): number {
  if (guesserPoints.length === 0) return 0
  return Math.round(guesserPoints.reduce((a, b) => a + b, 0) / guesserPoints.length)
}

export function psychicOrder(playerIds: string[], laps: number): string[] {
  return Array.from({ length: laps }, () => playerIds).flat()
}

export function freePlayWinners(totals: Record<string, number>): string[] {
  const top = Math.max(...Object.values(totals))
  return Object.keys(totals).filter((id) => totals[id] === top)
}
```

`dial-geometry.ts`:

```ts
const RAD = Math.PI / 180

export function valueToAngle(v: number): number {
  return 180 - v * 1.8
}

export function valueToPoint(cx: number, cy: number, r: number, v: number) {
  const a = valueToAngle(v) * RAD
  return { x: cx + r * Math.cos(a), y: cy - r * Math.sin(a) }
}

export function pointToValue(cx: number, cy: number, px: number, py: number): number {
  const angle = Math.atan2(cy - py, px - cx) / RAD // (-180, 180]
  if (angle < 0) return px < cx ? 0 : 100
  return Math.min(100, Math.max(0, (180 - angle) / 1.8))
}

export function wedgePath(cx: number, cy: number, rInner: number, rOuter: number, from: number, to: number): string {
  const a = valueToPoint(cx, cy, rOuter, from)
  const b = valueToPoint(cx, cy, rOuter, to)
  const c = valueToPoint(cx, cy, rInner, to)
  const d = valueToPoint(cx, cy, rInner, from)
  return `M${a.x} ${a.y} A${rOuter} ${rOuter} 0 0 1 ${b.x} ${b.y} L${c.x} ${c.y} A${rInner} ${rInner} 0 0 0 ${d.x} ${d.y} Z`
}
```

`crank.ts`:

```ts
export const GEAR_RATIO = 2.5
export const CLICK_STEP = Math.PI / 12

export function unwrapDelta(prev: number, next: number): number {
  let d = next - prev
  while (d > Math.PI) d -= 2 * Math.PI
  while (d <= -Math.PI) d += 2 * Math.PI
  return d
}

export function clicksCrossed(prevAngle: number, nextAngle: number, step: number): number {
  return Math.abs(Math.floor(nextAngle / step) - Math.floor(prevAngle / step))
}
```

Note: the `wedgePath` arcs go from `from` to `to` where larger values are further clockwise on screen (value 0 is the left end), so the outer arc uses sweep-flag 1 and the inner arc returns with sweep-flag 0.

```sql
  dial_deadline timestamptz,

  created_at timestamptz not null default now(),
  unique (room_id, number)
);

create table round_secrets (
  round_id uuid primary key references rounds on delete cascade,
  target numeric not null check (target between 10 and 90)
);

create table guesses (
  round_id uuid not null references rounds on delete cascade,

      or exists (select 1 from seats where room_id = p_room and session_id = auth.uid())
      or exists (select 1 from viewers where room_id = p_room and session_id = auth.uid());

create policy rounds_read on rounds for select using (is_room_member(room_id));

  exists (
    select 1 from rounds r
    where r.id = round_secrets.round_id
      and is_room_member(r.room_id)
      and (
        r.phase = 'reveal'

      and (
        m.mode = 'teams'

      )
  )
);
```

```sql
create function _require_host(p_room uuid) returns void
language plpgsql security definer set search_path = public as $$
begin
  if not exists (select 1 from rooms where id = p_room and host_session = auth.uid()) then
    raise exception 'host only';
  end if;
end $$;

create function create_room(p_mode text, p_winning int, p_laps int, p_timer int) returns text
language plpgsql security definer set search_path = public as $$
declare
  alphabet constant text := 'ABCDEFGHJKLMNPQRSTUVWXYZ';
  v_code text;
begin
  if auth.uid() is null then raise exception 'sign in first'; end if;
  loop
    v_code := '';
    for i in 1..4 loop
      v_code := v_code || substr(alphabet, 1 + floor(random() * length(alphabet))::int, 1);
    end loop;
    exit when not exists (select 1 from rooms where code = v_code);
  end loop;
  insert into rooms (code, host_session, mode, winning_score, laps, timer_secs)
  values (v_code, auth.uid(), p_mode, coalesce(p_winning, 10), coalesce(p_laps, 2), p_timer);
  return v_code;
end $$;

create function join_room(p_code text, p_name text, p_color text, p_members text[]) returns uuid
language plpgsql security definer set search_path = public as $$
declare
  r rooms;
  v_seat uuid;
  v_count int;
  v_max int;
begin
  select * into r from rooms where code = upper(p_code) for update;
  if not found then raise exception 'room not found'; end if;
  select id into v_seat from seats where room_id = r.id and session_id = auth.uid();
  if found then return v_seat; end if; -- rejoin
  if r.status <> 'lobby' then raise exception 'game already started'; end if;
  select count(*) into v_count from seats where room_id = r.id;
  v_max := case r.mode when 'teams' then 6 else 12 end;
  if v_count >= v_max then raise exception 'room is full'; end if;
  insert into seats (room_id, session_id, position, name, color, members)
  values (r.id, auth.uid(), v_count, p_name, p_color,
          case r.mode when 'free' then array[p_name] else coalesce(p_members, '{}') end)
  returning id into v_seat;
  update rooms set last_active = now() where id = r.id;
  return v_seat;
end $$;

create function join_as_viewer(p_code text) returns uuid
language plpgsql security definer set search_path = public as $$
declare v_room uuid;
begin
  select id into v_room from rooms where code = upper(p_code);
  if v_room is null then raise exception 'room not found'; end if;
  insert into viewers (room_id, session_id) values (v_room, auth.uid()) on conflict do nothing;
  return v_room;
end $$;

create function update_seat(p_seat uuid, p_name text, p_members text[]) returns void
language plpgsql security definer set search_path = public as $$
begin
  update seats set name = p_name, members = p_members
  where id = p_seat and session_id = auth.uid()
    and exists (select 1 from rooms where id = seats.room_id and status = 'lobby');
end $$;

create function kick_seat(p_seat uuid) returns void
language plpgsql security definer set search_path = public as $$
declare v_room uuid;
begin
  select room_id into v_room from seats where id = p_seat;
  perform _require_host(v_room);
  delete from seats where id = p_seat;
end $$;

create function transfer_host(p_room uuid, p_session uuid) returns void
language plpgsql security definer set search_path = public as $$
begin
  perform _require_host(p_room);
  update rooms set host_session = p_session where id = p_room;
end $$;

create function new_round(p_room uuid) returns uuid
language plpgsql security definer set search_path = public as $$
declare
  r rooms;
  v_seat seats;
  v_number int;
  v_round uuid;
  v_options int[];
  v_psychic text;
begin
  select * into r from rooms where id = p_room;
  select coalesce(max(number), 0) + 1 into v_number from rounds where room_id = p_room;
  if r.mode = 'teams' then
    select * into v_seat from seats where id = (r.game->>'active')::uuid;
  else
    select * into v_seat from seats where room_id = p_room
      order by position offset ((v_number - 1) % (select count(*) from seats where room_id = p_room)) limit 1;
  end if;
  v_psychic := v_seat.members[(v_seat.turns_taken % greatest(cardinality(v_seat.members), 1)) + 1];
  update seats set turns_taken = turns_taken + 1 where id = v_seat.id;
  select array_agg(id) into v_options from (
    select id from cards where room_id is null or room_id = p_room order by random() limit 2
  ) c;
  insert into rounds (room_id, number, seat_id, psychic_name, card_options)
  values (p_room, v_number, v_seat.id, coalesce(v_psychic, v_seat.name), v_options)
  returning id into v_round;
  return v_round;
end $$;

create function start_game(p_room uuid) returns void
language plpgsql security definer set search_path = public as $$
declare
  r rooms;
  v_count int;
  v_first uuid;
begin
  perform _require_host(p_room);
  select * into r from rooms where id = p_room for update;
  select count(*) into v_count from seats where room_id = p_room;
  if r.status <> 'lobby' then raise exception 'already started'; end if;
  if r.mode = 'teams' and v_count < 2 then raise exception 'need at least 2 teams'; end if;
  if r.mode = 'free' and v_count < 3 then raise exception 'need at least 3 players'; end if;
  if r.mode = 'teams' and exists (select 1 from seats where room_id = p_room and cardinality(members) = 0) then
    raise exception 'every team needs at least one member';
  end if;
  select id into v_first from seats where room_id = p_room order by position limit 1;
  update rooms set status = 'playing', last_active = now(),
    game = jsonb_build_object('phase', 'playing', 'active', v_first, 'winners', '[]'::jsonb)
  where id = p_room;
  perform new_round(p_room);
end $$;

-- Snapshot for clients: never includes the target; guesses follow RLS.
create function get_room_state(p_code text) returns jsonb
language plpgsql stable security invoker set search_path = public as $$
declare
  r rooms;
  v_round rounds;
begin
  select * into r from rooms where code = upper(p_code);
  if not found then return null; end if;
  select * into v_round from rounds where room_id = r.id order by number desc limit 1;
  return jsonb_build_object(
    'room', to_jsonb(r) - 'host_session',
    'is_host', r.host_session = auth.uid(),
    'my_seat', (select id from seats where room_id = r.id and session_id = auth.uid()),
    'seats', coalesce((select jsonb_agg(to_jsonb(s) - 'session_id' order by s.position) from seats s where s.room_id = r.id), '[]'),
    'round', to_jsonb(v_round),
    'cards', coalesce((select jsonb_agg(to_jsonb(c)) from cards c where c.id = any(v_round.card_options) or c.id = v_round.card_id), '[]'),
    'guesses', coalesce((select jsonb_agg(to_jsonb(g)) from guesses g where g.round_id = v_round.id), '[]')
  );
end $$;

revoke all on function new_round(uuid) from public, anon, authenticated;
```

- `result` jsonb shape written on reveal: `{ "target": n, "dial_points": n, "entries": [{ "seat_id": uuid, "dial": n|null, "points": n, "lr": "left"|"right"|null, "lr_correct": bool }], "psychic_points": n }`

```sql
create function band_points(p_target numeric, p_dial numeric) returns int
language sql immutable as $$
  select case
    when abs(p_target - p_dial) <= 2 then 4
    when abs(p_target - p_dial) <= 6 then 3
    when abs(p_target - p_dial) <= 10 then 2
    else 0 end;
$$;

create function _next_seat(p_room uuid, p_seat uuid) returns uuid
language sql stable as $$
  select coalesce(
    (select id from seats where room_id = p_room and position > (select position from seats where id = p_seat) order by position limit 1),
    (select id from seats where room_id = p_room order by position limit 1));
$$;

create function _sd_state(p_room uuid, p_leaders uuid[]) returns jsonb
language sql stable as $$
  select jsonb_build_object(
    'phase', 'sudden_death',
    'sd_tied', to_jsonb(p_leaders),
    'sd_pending', to_jsonb(p_leaders),
    'active', p_leaders[1],
    'sd_start', (select jsonb_object_agg(id::text, score) from seats where room_id = p_room));
$$;

-- Mirrors src/game-engine/team.ts applyTurn.
create function apply_team_turn(p_room uuid, p_dial_pts int, p_bonus uuid[]) returns void
language plpgsql security definer set search_path = public as $$
declare
  r rooms; g jsonb; v_active uuid; v_phase text;
  v_active_score int; v_best_other int; v_top int;
  v_leaders uuid[]; v_pending uuid[];
begin
  select * into r from rooms where id = p_room for update;
  g := r.game; v_active := (g->>'active')::uuid; v_phase := g->>'phase';

  update seats set score = score + p_dial_pts where id = v_active;
  update seats set score = score + 1
   where room_id = p_room and id = any(p_bonus) and id <> v_active
     and (v_phase <> 'sudden_death' or id::text in (select jsonb_array_elements_text(g->'sd_tied')));

  if v_phase = 'sudden_death' then
    v_pending := array(select x::uuid from jsonb_array_elements_text(g->'sd_pending') x where x::uuid <> v_active);
    if cardinality(v_pending) > 0 then
      g := g || jsonb_build_object('sd_pending', to_jsonb(v_pending), 'active', v_pending[1]);
    else
      with gains as (
        select s.id, s.score - (g->'sd_start'->>(s.id::text))::int as gain
        from seats s
        where s.room_id = p_room and s.id::text in (select jsonb_array_elements_text(g->'sd_tied'))
      )
      select array_agg(id) into v_leaders from gains where gain = (select max(gain) from gains);
      if cardinality(v_leaders) = 1 then
        g := g || jsonb_build_object('phase', 'finished', 'winners', to_jsonb(v_leaders), 'sd_pending', '[]'::jsonb);
      else
        g := g || _sd_state(p_room, v_leaders);
      end if;
    end if;
  else
    select score into v_active_score from seats where id = v_active;
    select max(score) into v_best_other from seats where room_id = p_room and id <> v_active;
    if p_dial_pts = 4 and v_active_score < v_best_other then
      null; -- catch-up turn: same team goes again
    else
      select max(score) into v_top from seats where room_id = p_room;
      if v_top >= r.winning_score then
        select array_agg(id) into v_leaders from seats where room_id = p_room and score = v_top;
        if cardinality(v_leaders) = 1 then
          g := g || jsonb_build_object('phase', 'finished', 'winners', to_jsonb(v_leaders));
        else
          g := g || _sd_state(p_room, v_leaders);
        end if;
      else
        g := g || jsonb_build_object('active', _next_seat(p_room, v_active));
      end if;
    end if;
  end if;

  update rooms set game = g,
    status = case when g->>'phase' = 'finished' then 'finished' else status end,
    last_active = now()
  where id = p_room;
end $$;

create function _round_for_active(p_round uuid, p_phase text) returns rounds
language plpgsql security definer set search_path = public as $$
declare v rounds;
begin
  select r.* into v from rounds r join seats s on s.id = r.seat_id
   where r.id = p_round and s.session_id = auth.uid();
  if not found then raise exception 'not your turn'; end if;
  if v.phase <> p_phase then raise exception 'wrong phase: %', v.phase; end if;
  return v;
end $$;

create function pick_card(p_round uuid, p_card int) returns void
language plpgsql security definer set search_path = public as $$
declare v rounds;
begin
  v := _round_for_active(p_round, 'card_pick');
  if not (p_card = any(v.card_options)) then raise exception 'card not offered'; end if;
  insert into round_secrets (round_id, target) values (p_round, round((10 + random() * 80)::numeric, 2));
  update rounds set card_id = p_card, phase = 'spin' where id = p_round;
end $$;

create function finish_spin(p_round uuid) returns void
language plpgsql security definer set search_path = public as $$
begin
  perform _round_for_active(p_round, 'spin');
  update rounds set phase = 'clue' where id = p_round;
end $$;

create function submit_clue(p_round uuid, p_clue text) returns void
language plpgsql security definer set search_path = public as $$
declare v rounds; v_timer int;
begin
  v := _round_for_active(p_round, 'clue');
  select timer_secs into v_timer from rooms where id = v.room_id;
  update rounds set clue = nullif(trim(p_clue), ''), phase = 'dial',
    dial_deadline = case when v_timer is null then null else now() + make_interval(secs => v_timer) end
  where id = p_round;
end $$;

$$;

create function reveal_round(p_round uuid) returns void
language plpgsql security definer set search_path = public as $$
declare
  v rounds; r rooms; v_target numeric;
  v_entries jsonb; v_dial_pts int := 0; v_bonus uuid[]; v_psychic_pts int := 0;
begin
  select * into v from rounds where id = p_round for update;
  if v.phase = 'reveal' then return; end if;
  select * into r from rooms where id = v.room_id;
  if auth.uid() is not null and not (
    r.host_session = auth.uid()
    or exists (select 1 from seats where room_id = r.id and session_id = auth.uid())
  ) then raise exception 'not allowed'; end if;
  select target into v_target from round_secrets where round_id = p_round;

  if r.mode = 'teams' then
    select coalesce(band_points(v_target, g.dial), 0) into v_dial_pts
      from guesses g where g.round_id = p_round and g.seat_id = v.seat_id;
    v_dial_pts := coalesce(v_dial_pts, 0);
    select array_agg(g.seat_id) into v_bonus
      from guesses g join guesses ag on ag.round_id = g.round_id and ag.seat_id = v.seat_id
     where g.round_id = p_round and g.seat_id <> v.seat_id and ag.dial is not null
       and g.lr is not null and v_target <> ag.dial
       and g.lr = case when v_target < ag.dial then 'left' else 'right' end;
    select jsonb_agg(jsonb_build_object(
        'seat_id', g.seat_id, 'dial', g.dial, 'lr', g.lr,
        'points', case when g.seat_id = v.seat_id then v_dial_pts else 0 end,
        'lr_correct', g.seat_id = any(coalesce(v_bonus, '{}'))))
      into v_entries from guesses g where g.round_id = p_round;
    update rounds set phase = 'reveal', result = jsonb_build_object(
      'target', v_target, 'dial_points', v_dial_pts, 'entries', coalesce(v_entries, '[]'), 'psychic_points', 0)
    where id = p_round;
    perform apply_team_turn(r.id, v_dial_pts, coalesce(v_bonus, '{}'));
  else
    select jsonb_agg(jsonb_build_object('seat_id', g.seat_id, 'dial', g.dial, 'lr', null,
             'points', band_points(v_target, g.dial), 'lr_correct', false)),
           coalesce(round(avg(band_points(v_target, g.dial))), 0)::int
      into v_entries, v_psychic_pts from guesses g where g.round_id = p_round and g.dial is not null;
    update seats set score = score + band_points(v_target, g.dial)
      from guesses g where g.round_id = p_round and g.seat_id = seats.id and g.dial is not null;
    update seats set score = score + v_psychic_pts where id = v.seat_id;
    update rounds set phase = 'reveal', result = jsonb_build_object(
      'target', v_target, 'dial_points', 0, 'entries', coalesce(v_entries, '[]'), 'psychic_points', v_psychic_pts)
    where id = p_round;
    update rooms set last_active = now() where id = r.id;
  end if;
end $$;

create function lock_dial(p_round uuid, p_dial numeric) returns void
language plpgsql security definer set search_path = public as $$
declare v rounds; r rooms; v_seat seats; v_needed int; v_locked int;
begin
  select * into v from rounds where id = p_round for update;
  select * into r from rooms where id = v.room_id;
  select * into v_seat from seats where room_id = r.id and session_id = auth.uid();
  if v_seat.id is null then raise exception 'not seated'; end if;
  if v.phase <> 'dial' then raise exception 'wrong phase: %', v.phase; end if;
  if r.mode = 'teams' and v_seat.id <> v.seat_id then raise exception 'only the active team sets the dial'; end if;
  if r.mode = 'free' and v_seat.id = v.seat_id then raise exception 'the psychic does not guess'; end if;
  insert into guesses (round_id, seat_id, dial) values (p_round, v_seat.id, p_dial)
    on conflict (round_id, seat_id) do update set dial = excluded.dial;
  if r.mode = 'teams' then
    update rounds set phase = 'lr' where id = p_round;
  else
    select count(*) into v_needed from seats where room_id = r.id and id <> v.seat_id;
    select count(*) into v_locked from guesses where round_id = p_round and dial is not null;
    if v_locked >= v_needed then perform reveal_round(p_round); end if;
  end if;
end $$;

create function lock_lr(p_round uuid, p_guess text) returns void
language plpgsql security definer set search_path = public as $$
declare v rounds; r rooms; v_seat seats; v_needed int; v_locked int;
begin
  select * into v from rounds where id = p_round for update;
  select * into r from rooms where id = v.room_id;
  select * into v_seat from seats where room_id = r.id and session_id = auth.uid();
  if r.mode <> 'teams' or v.phase <> 'lr' or v_seat.id is null or v_seat.id = v.seat_id then
    raise exception 'cannot guess left/right now';
  end if;
  insert into guesses (round_id, seat_id, lr) values (p_round, v_seat.id, p_guess)
    on conflict (round_id, seat_id) do update set lr = excluded.lr;
  select count(*) into v_needed from seats where room_id = r.id and id <> v.seat_id;
  select count(*) into v_locked from guesses where round_id = p_round and seat_id <> v.seat_id and lr is not null;
  if v_locked >= v_needed then perform reveal_round(p_round); end if;
end $$;

create function next_turn(p_room uuid) returns void
language plpgsql security definer set search_path = public as $$
declare r rooms; v_rounds int; v_seats int; v_top int; v_winners uuid[];
begin
  perform _require_host(p_room);
  select * into r from rooms where id = p_room for update;
  if r.game->>'phase' = 'finished' then return; end if;
  if not exists (select 1 from rounds where room_id = p_room and phase = 'reveal'
                 and number = (select max(number) from rounds where room_id = p_room)) then
    raise exception 'round not revealed yet';
  end if;
  if r.mode = 'free' then
    select count(*) into v_rounds from rounds where room_id = p_room;
    select count(*) into v_seats from seats where room_id = p_room;
    if v_rounds >= r.laps * v_seats then
      select max(score) into v_top from seats where room_id = p_room;
      select array_agg(id) into v_winners from seats where room_id = p_room and score = v_top;
      update rooms set status = 'finished', game = game || jsonb_build_object('phase', 'finished', 'winners', to_jsonb(v_winners))
      where id = p_room;
      return;
    end if;
  end if;
  perform new_round(p_room);
end $$;

create function rematch(p_room uuid) returns void
language plpgsql security definer set search_path = public as $$
declare v_first uuid;
begin
  perform _require_host(p_room);
  delete from rounds where room_id = p_room;
  update seats set score = 0, turns_taken = 0 where room_id = p_room;
  select id into v_first from seats where room_id = p_room order by position limit 1;
  update rooms set status = 'playing', last_active = now(),
    game = jsonb_build_object('phase', 'playing', 'active', v_first, 'winners', '[]'::jsonb)
  where id = p_room;
  perform new_round(p_room);
end $$;

grant execute on function
  create_room, join_room, join_as_viewer, update_seat, kick_seat, transfer_host, start_game,
  get_room_state, pick_card, finish_spin, submit_clue, get_target, reveal_round, lock_dial,
  lock_lr, next_turn, rematch
to authenticated;
revoke all on function apply_team_turn(uuid, int, uuid[]), _sd_state(uuid, uuid[]), _next_seat(uuid, uuid),
  _round_for_active(uuid, text), _require_host(uuid), band_points(numeric, numeric) from public, anon, authenticated;
```

`throttle.ts`:

```ts
export function throttle<T extends unknown[]>(fn: (...a: T) => void, ms: number) {
  let last = 0
  let timer: ReturnType<typeof setTimeout> | undefined
  let pending: T | undefined
  return (...args: T) => {
    const now = Date.now()
    if (now - last >= ms) {
      last = now
      fn(...args)
      return
    }
    pending = args
    if (!timer) {
      timer = setTimeout(() => {
        timer = undefined
        last = Date.now()
        if (pending) fn(...pending)
        pending = undefined
      }, ms - (now - last))
    }
  }
}
```

```ts
export function useRoom(code: string) {
  const [state, setState] = useState<RoomState | null>(null)
  const [error, setError] = useState<string | null>(null)
  const timer = useRef<ReturnType<typeof setTimeout>>()

  const refresh = useCallback(() => {
    clearTimeout(timer.current)
    timer.current = setTimeout(async () => {
      try {
        const s = await api.getRoomState(code)
        if (!s) setError('Room not found')
        else { setState(s); setError(null) }
      } catch (e) { setError((e as Error).message) }
    }, 50)
  }, [code])

  useEffect(() => {
    refresh()
    const channel = supabase.channel(`db:${code}`)
    for (const table of ['rooms', 'seats', 'rounds', 'guesses']) {
      channel.on('postgres_changes', { event: '*', schema: 'public', table }, refresh)
    }
    channel.subscribe()
    return () => { clearTimeout(timer.current); supabase.removeChannel(channel) }
  }, [code, refresh])

  return { state, error, refresh }
}
```

```ts
import { useSyncExternalStore } from 'react'

export interface Settings { muted: boolean; volume: number }
interface Store { getItem(k: string): string | null; setItem(k: string, v: string): void }
const KEY = 'dialed.settings'
const DEFAULTS: Settings = { muted: false, volume: 0.7 }

export function createSettings(storage: Store) {
  let current: Settings = { ...DEFAULTS }
  try {
    const raw = storage.getItem(KEY)
    if (raw) current = { ...DEFAULTS, ...JSON.parse(raw) }
  } catch { /* storage unavailable: keep defaults */ }
  const listeners = new Set<() => void>()
  return {
    get: () => current,
    set(patch: Partial<Settings>) {
      current = { ...current, ...patch, volume: Math.min(1, Math.max(0, patch.volume ?? current.volume)) }
      try { storage.setItem(KEY, JSON.stringify(current)) } catch { /* ignore */ }
      listeners.forEach((l) => l())
    },
    subscribe(fn: () => void) { listeners.add(fn); return () => listeners.delete(fn) },
  }
}

const safeStorage: Store = {
  getItem: (k) => { try { return localStorage.getItem(k) } catch { return null } },
  setItem: (k, v) => { try { localStorage.setItem(k, v) } catch { /* ignore */ } },
}
export const settings = createSettings(safeStorage)
export const useSettings = () => useSyncExternalStore(settings.subscribe, settings.get)
```

```ts
import { settings } from './settings'

let ctx: AudioContext | undefined
let master: GainNode | undefined
const live = new Set<AudioScheduledSourceNode>()

function ac() {
  if (!ctx) {
    ctx = new AudioContext()
    master = ctx.createGain()
    master.connect(ctx.destination)
  }
  master!.gain.value = settings.get().muted ? 0 : settings.get().volume
  return ctx
}

function blip(freq: number, dur: number, type: OscillatorType = 'square', gain = 0.2, when = 0, slideTo?: number) {
  const c = ac()
  const t = c.currentTime + when
  const o = c.createOscillator()
  const g = c.createGain()
  o.type = type
  o.frequency.setValueAtTime(freq, t)
  if (slideTo) o.frequency.exponentialRampToValueAtTime(slideTo, t + dur)
  g.gain.setValueAtTime(gain, t)
  g.gain.exponentialRampToValueAtTime(0.0001, t + dur)
  o.connect(g).connect(master!)
  o.start(t)
  o.stop(t + dur + 0.02)
  live.add(o)
  o.onended = () => live.delete(o)
}

function noise(dur: number, from: number, to: number, gain = 0.25) {
  const c = ac()
  const buf = c.createBuffer(1, Math.ceil(c.sampleRate * dur), c.sampleRate)
  const data = buf.getChannelData(0)
  for (let i = 0; i < data.length; i++) data[i] = Math.random() * 2 - 1
  const src = c.createBufferSource()
  src.buffer = buf
  const f = c.createBiquadFilter()
  f.type = 'bandpass'
  f.frequency.setValueAtTime(from, c.currentTime)
  f.frequency.exponentialRampToValueAtTime(to, c.currentTime + dur)
  const g = c.createGain()
  g.gain.setValueAtTime(gain, c.currentTime)
  g.gain.exponentialRampToValueAtTime(0.0001, c.currentTime + dur)
  src.connect(f).connect(g).connect(master!)
  src.start()
  live.add(src)
  src.onended = () => live.delete(src)
}

export const audio = {
  unlock() { void ac().resume() },
  /** Dial tick; `pitch` 0..1 shifts the note with needle position. */
  tick(pitch = 0.5) { blip(500 + pitch * 500, 0.04, 'square', 0.12) },
  /** Crank ratchet click; faster cranking -> higher and shorter. */
  ratchet(speed: number) {
    const s = Math.min(1, Math.max(0, speed))
    blip(700 + s * 900, 0.03 - s * 0.01, 'square', 0.14)
  },
  clunk() { blip(140, 0.18, 'triangle', 0.5, 0, 60); noise(0.08, 400, 200, 0.2) },
  swoosh() { noise(0.45, 300, 3200, 0.22) },
  drumroll(ms: number) {
    const hits = Math.max(4, Math.floor(ms / 45))
    for (let i = 0; i < hits; i++) blip(120 + i * 4, 0.05, 'triangle', 0.08 + (i / hits) * 0.18, (i * ms) / hits / 1000)
  },
  chime(points: number) {
    const notes = [523.25, 659.25, 783.99, 1046.5]
    const n = Math.max(0, Math.min(4, points))
    for (let i = 0; i < n; i++) blip(notes[i], 0.35, 'sine', 0.22, i * 0.09)
    if (n === 0) blip(200, 0.4, 'sawtooth', 0.12, 0, 90)
  },
  timerTick() { blip(900, 0.05, 'square', 0.1) },
  stopAll() { live.forEach((n) => { try { n.stop() } catch { /* already stopped */ } }); live.clear() },
```

- [ ] **Step 3: Implement the crank**

The crank handle sits on a circular track. While the pointer is down, take the angle of the pointer around the track centre, compute `delta = unwrapDelta(prev, next)`, add `delta` to `crankAngle` and `delta * GEAR_RATIO` (in degrees) to the wheel `angle` motion value, play a ratchet click per `CLICK_STEP` crossed (pitch from speed), and track the angular velocity (rad/s) over the last few pointer events. On release, hand the velocity to Motion's inertia animation on the wheel; when it ends, check the total rotation.

```tsx
import { animate, useMotionValue } from 'motion/react'
import { useRef, useState } from 'react'
import { audio } from '@/lib/audio'
import { CLICK_STEP, GEAR_RATIO, clicksCrossed, unwrapDelta } from '@/lib/crank'
import { haptics, prefersReducedMotion } from '@/lib/haptics'
import { SpinWheel } from './SpinWheel'

const DEG = 180 / Math.PI

export function Crank({ onDone, minTurns = 2 }: { onDone: () => void; minTurns?: number }) {
  const wheel = useMotionValue(0)
  const [knob, setKnob] = useState(0) // radians, for the knob position
  const track = useRef<HTMLDivElement>(null)
  const last = useRef<{ a: number; t: number } | null>(null)
  const velocity = useRef(0) // rad/s
  const travelled = useRef(0) // total wheel degrees, absolute
  const crankTotal = useRef(0)
  const finished = useRef(false)

  const finish = () => { if (!finished.current) { finished.current = true; haptics.thud(); audio.clunk(); onDone() } }
  const enough = () => travelled.current >= minTurns * 360

  const turn = (deltaRad: number, dt: number) => {
    const prevTotal = crankTotal.current
    crankTotal.current += deltaRad
    const wheelDeg = deltaRad * GEAR_RATIO * DEG
    wheel.set(wheel.get() + wheelDeg)
    travelled.current += Math.abs(wheelDeg)
    const speed = Math.min(1, Math.abs(deltaRad / Math.max(dt, 0.001)) / 12)
    const n = clicksCrossed(prevTotal, crankTotal.current, CLICK_STEP)
    for (let i = 0; i < Math.min(n, 3); i++) { audio.ratchet(speed); if (i === 0) haptics.tick() }
  }

  const angleOf = (e: React.PointerEvent) => {
    const r = track.current!.getBoundingClientRect()
    return Math.atan2(e.clientY - (r.top + r.height / 2), e.clientX - (r.left + r.width / 2))
  }

  const onDown = (e: React.PointerEvent) => {
    audio.unlock()
    e.currentTarget.setPointerCapture(e.pointerId)
    last.current = { a: angleOf(e), t: performance.now() }
  }
  const onMove = (e: React.PointerEvent) => {
    if (!last.current || finished.current) return
    const a = angleOf(e)
    const t = performance.now()
    const dt = (t - last.current.t) / 1000
    const d = unwrapDelta(last.current.a, a)
    velocity.current = d / Math.max(dt, 0.001)
    last.current = { a, t }
    setKnob(a)
    turn(d, dt)
  }
  const onUp = () => {
    if (!last.current) return
    last.current = null
    const v = velocity.current * GEAR_RATIO * DEG
    animate(wheel, wheel.get() + v * 0.6, {
      type: 'inertia', velocity: v, power: 0.6, timeConstant: 500,
      onUpdate: (w) => {
        const dW = Math.abs(w - lastWheel.current); lastWheel.current = w
        travelled.current += dW
        const n = clicksCrossed((w - dW) / DEG, w / DEG, CLICK_STEP / GEAR_RATIO)
        if (n > 0) audio.ratchet(Math.min(1, dW / 20))
      },
      onComplete: () => { if (enough()) finish() },
    })
  }
  const lastWheel = useRef(0)

  const autoSpin = () => {
    audio.unlock()
    if (prefersReducedMotion()) { travelled.current = minTurns * 360; finish(); return }
    const to = wheel.get() + 360 * (minTurns + 1.3)
    lastWheel.current = wheel.get()
    animate(wheel, to, {
      duration: 2.6, ease: [0.1, 0.7, 0.2, 1],
      onUpdate: (w) => {
        const dW = Math.abs(w - lastWheel.current); lastWheel.current = w
        travelled.current += dW
        if (clicksCrossed((w - dW) / DEG, w / DEG, CLICK_STEP / GEAR_RATIO) > 0) audio.ratchet(Math.min(1, dW / 12))
      },
      onComplete: finish,
    })
  }

  return (
    <div className="flex flex-col items-center gap-4">
      <SpinWheel angle={wheel} />
      <div ref={track} className="relative size-40 rounded-full border-8 border-teal-800 bg-teal-950/40 touch-none"
        onPointerDown={onDown} onPointerMove={onMove} onPointerUp={onUp} onPointerCancel={onUp} data-testid="crank-track">
        <div className="absolute size-12 rounded-full bg-orange-500 shadow-lg border-4 border-orange-700"
          style={{ left: `calc(50% + ${Math.cos(knob) * 56}px - 24px)`, top: `calc(50% + ${Math.sin(knob) * 56}px - 24px)` }} />
        <div className="absolute inset-0 grid place-items-center text-xs text-teal-100/70 pointer-events-none">Crank me</div>
      </div>
      <button className="text-sm underline" onClick={autoSpin} data-testid="spin-for-me">Spin for me</button>
    </div>
  )
}
```

```ts
import type { RoundResult } from './types'

export interface RevealStep {
  at: number
  kind: 'clunk' | 'lr' | 'drumroll' | 'open' | 'glow' | 'points' | 'bonus'
  seatId?: string
  points?: number
}

export function buildRevealTimeline(result: RoundResult, mode: 'teams' | 'free'): RevealStep[] {
  const steps: RevealStep[] = [{ at: 0, kind: 'clunk' }]
  const lr = result.entries.filter((e) => e.lr)
  lr.forEach((e, i) => steps.push({ at: 250 + i * 150, kind: 'lr', seatId: e.seat_id }))
  const drumStart = 250 + lr.length * 150
  steps.push({ at: drumStart, kind: 'drumroll' })
  const open = drumStart + 900
  steps.push({ at: open, kind: 'open' })
  steps.push({ at: open + 350, kind: 'glow' })
  const scored = result.entries.filter((e) => e.points > 0)
  scored.forEach((e, i) => steps.push({ at: open + 700 + i * 200, kind: 'points', seatId: e.seat_id, points: e.points }))
  const bonusStart = open + 700 + Math.max(scored.length, 1) * 200 + 200
  if (mode === 'teams') {
    result.entries.filter((e) => e.lr_correct).forEach((e, i) =>
      steps.push({ at: bonusStart + i * 180, kind: 'bonus', seatId: e.seat_id, points: 1 }))
  }
  return steps
}
```

```sql
  select array_agg(id) into v_winners from seats where room_id = p_room and score = v_top;
  update rooms set status = 'finished',
    game = game || jsonb_build_object('phase', 'finished', 'winners', to_jsonb(v_winners))
  where id = p_room;
end $$;
grant execute on function end_game(uuid) to authenticated;
```

Add it to `0003_game_rpcs.sql` with `api.endGame`.
